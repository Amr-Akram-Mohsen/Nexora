CREATE OR REPLACE FUNCTION slugify(input TEXT)
RETURNS TEXT AS $$
BEGIN
    RETURN lower(regexp_replace(trim(input), '[^a-zA-Z0-9]+', '-', 'g'));
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION get_or_create_category(p_name TEXT)
RETURNS INT AS $$
DECLARE v_id INT;
BEGIN

    SELECT id INTO v_id
    FROM categories
    WHERE lower(name) = lower(p_name);

    IF v_id IS NOT NULL THEN
        RETURN v_id;
    END IF;

    INSERT INTO categories (name, slug)
    VALUES (p_name, slugify(p_name))
    RETURNING id INTO v_id;

    RETURN v_id;

END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION get_or_create_brand(p_name TEXT)
RETURNS INT AS $$
DECLARE v_id INT;
BEGIN

    SELECT id INTO v_id
    FROM brands
    WHERE lower(name) = lower(p_name);

    IF v_id IS NOT NULL THEN
        RETURN v_id;
    END IF;

    INSERT INTO brands (name, slug)
    VALUES (p_name, slugify(p_name))
    RETURNING id INTO v_id;

    RETURN v_id;

END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION create_item_with_store_link(
    p_name TEXT,
    p_description TEXT,
    p_category_name TEXT,
    p_brand_name TEXT,
    p_price NUMERIC,
    p_currency TEXT,
    p_store_id INT,
    p_affiliate_url TEXT,
    p_original_url TEXT
)
RETURNS INT AS $$
DECLARE
    v_item_id INT;
    v_variant_id INT;
    v_category_id INT;
    v_brand_id INT;
BEGIN

    -- 1. resolve category + brand (auto create if missing)
    v_category_id := get_or_create_category(p_category_name);
    v_brand_id := get_or_create_brand(p_brand_name);

    -- 2. insert item
    INSERT INTO items (
        name,
        slug,
        description,
        category_id,
        brand_id,
        item_type,
        created_at
    )
    VALUES (
        p_name,
        slugify(p_name),
        p_description,
        v_category_id,
        v_brand_id,
        'electronics',
        NOW()
    )
    RETURNING id INTO v_item_id;

    -- 3. insert default variant
    INSERT INTO item_variants (
        item_id,
        price,
        currency,
        is_default,
        created_at
    )
    VALUES (
        v_item_id,
        p_price,
        p_currency,
        TRUE,
        NOW()
    )
    RETURNING id INTO v_variant_id;

    -- 4. insert store link
    INSERT INTO item_store_links (
        variant_id,
        store_id,
        original_url,
        affiliate_url,
        price,
        currency,
        is_active,
        created_at
    )
    VALUES (
        v_variant_id,
        p_store_id,
        p_original_url,
        p_affiliate_url,
        p_price,
        p_currency,
        TRUE,
        NOW()
    );

    RETURN v_item_id;

END;
$$ LANGUAGE plpgsql;
